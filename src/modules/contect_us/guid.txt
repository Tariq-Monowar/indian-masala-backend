get all contect_us
______________GET
{{_baseUrl}}/api/contect_us/get

header: {
  - token: Admin
}

query: {
  - cursor={{id}}
  - limit={{limit}}
  - started_date={{YYYY-MM-DD}}
  - end_date={{YYYY-MM-DD}}
}

body: {
}

note:
  - send jwt in token header
  - cursor is the last contect_us id from the previous page
  - limit defaults to 20 and max is 50
  - do not send cursor on the first page
  - started_date and end_date filter createdAt
  - end_date includes that full day

Response: {
  {
    "success": true,
    "data": [
      {
        "id": "uuid",
        "first_name": "Tariq",
        "last_name": "Monowar",
        "email": "tariq@example.com",
        "phone": "01712345678",
        "message": "I want to book a table",
        "createdAt": "2026-09-12T10:00:00.000Z",
        "updatedAt": "2026-09-12T10:00:00.000Z"
      }
    ],
    "pagination": {
      "hasMore": false
    }
  }
}
