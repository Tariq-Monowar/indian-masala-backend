=================================
get all category
______________GET
{{_baseUrl}}/api/menu/category/get

header: {
}

query: {
  - cursor={{id}}
  - limit={{limit}}
  - search={{search}}
}

body: {
}

note:
  - cursor is the last category id from the previous page
  - limit defaults to 20
  - do not send cursor on the first page
  - search matches name

Response: {
  {
    "success": true,
    "data": [
      {
        "id": "uuid",
        "name": "Starters",
        "icon": "starter.png",
        "createdAt": "2026-09-12T10:00:00.000Z"
      }
    ],
    "pagination": {
      "hasMore": false
    }
  }
}
