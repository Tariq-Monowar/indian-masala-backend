create company_info
______________POST
{{_baseUrl}}/api/company_info/create

header: {
  - token: Admin
}

query: {
}

body: {
  - company_name: string
  - company_email: string
  - company_phone: string
  - address: string
  - city: string
  - country: string
  - location_label: string
}

note:
