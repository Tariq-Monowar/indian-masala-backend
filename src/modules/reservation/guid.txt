=================================
create reservation
______________POST
{{_baseUrl}}/api/reservation/create

header: {
}

query: {
}

body: {
  - number_of_guests: number
  - date: string
  - time: string
  - name: string
  - phone: string
  - email: string
  - description: string
  - status: pending | confirmed | cancelled | completed
}

note:
  - name, phone, date, time, number_of_guests are required
  - email and description are optional
  - date is YYYY-MM-DD
  - time is HH:MM or HH:MM:SS
  - status defaults to pending

=================================
get all reservation
______________GET
{{_baseUrl}}/api/reservation/get

header: {
  - token: Admin
}

query: {
  - cursor={{id}}
  - limit={{limit}}
  - search={{search}}
  - status={{status,status}}
  - started_date={{YYYY-MM-DD}}
  - end_date={{YYYY-MM-DD}}
}

body: {
}

note:
  - send jwt in token header
  - cursor is the last reservation id from the previous page
  - limit defaults to 20 and max is 50
  - do not send cursor on the first page
  - search matches name, phone, email, status, date, time, number_of_guests
  - status is pending, confirmed, cancelled, completed
  - status=pending,confirmed returns those statuses
  - started_date and end_date filter createdAt
  - end_date includes that full day

Response: {
  {
    "success": true,
    "data": [
      {
        "id": "uuid",
        "number_of_guests": 4,
        "date": "2026-09-20",
        "time": "18:30:00",
        "name": "Tariq Monowar",
        "phone": "01712345678",
        "email": "tariq@example.com",
        "status": "pending",
        "createdAt": "2026-09-12T10:00:00.000Z",
        "updatedAt": "2026-09-12T10:00:00.000Z"
      }
    ],
    "pagination": {
      "hasMore": false
    }
  }
}

}
