=================================
get dashboard stats
______________GET
{{_baseUrl}}/api/dashboard/get

header: {
  - token: Admin
}

query: {
  - period={{today|7days|month|year}}
}

body: {
}

note:
  - send jwt in token header
  - period defaults to today
  - today compares vs yesterday
  - 7days compares vs previous 7 days
  - month compares vs last calendar month
  - year compares vs last calendar year
  - change_percent is absolute number for UI like 10% vs yesterday
  - direction is up, down or flat

Response: {
  {
    "success": true,
    "data": {
      "period": "today",
      "compare_label": "vs yesterday",
      "orders": {
        "count": 25,
        "change_percent": 10,
        "direction": "up"
      },
      "reservations": {
        "count": 12,
        "change_percent": 6,
        "direction": "up"
      },
      "catering": {
        "count": 4,
        "change_percent": 8,
        "direction": "down"
      }
    }
  }
}
