=================================
create catering
______________POST
{{_baseUrl}}/api/catering/create

query: {
}

body: {
  - name: string
  - phone: string
  - email: string
  - event_type: string
  - number_of_guests: number
  - date: string
  - location: string
  - special_requirements: string
  - description: string
  - status: pending | confirmed | cancelled | completed
}

note:
  - name, phone, date, number_of_guests are required
  - email, event_type, location, special_requirements, description are optional
  - date is YYYY-MM-DD
  - status defaults to pending

=================================
get all catering
______________GET
{{_baseUrl}}/api/catering/get

query: {
  - cursor={{id}}
  - limit={{limit}}
  - search={{search}}
  - status={{status,status}}
  - started_date={{YYYY-MM-DD}}
  - end_date={{YYYY-MM-DD}}
}

body: {
}

note:
  - Authorization admin token
  - cursor is the last catering id from the previous page
  - limit defaults to 20 and max is 50
  - do not send cursor on the first page
  - search matches name, phone, email, event_type, location, status, date, number_of_guests
  - status is pending, confirmed, cancelled, completed
  - status=pending,confirmed returns those statuses
  - started_date and end_date filter createdAt
  - end_date includes that full day
  - description and special_requirements are not returned

Response: {
  {
    "success": true,
    "data": [
      {
        "id": "uuid",
        "name": "Tariq Monowar",
        "phone": "01712345678",
        "email": "tariq@example.com",
        "event_type": "Wedding",
        "number_of_guests": 80,
        "date": "2026-10-15",
        "location": "Dhaka",
        "status": "pending",
        "createdAt": "2026-09-12T10:00:00.000Z",
        "updatedAt": "2026-09-12T10:00:00.000Z"
      }
    ],
    "pagination": {
      "hasMore": false
    }
  }
}
